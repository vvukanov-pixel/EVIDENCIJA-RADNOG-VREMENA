/** * ISPRAVLJENA VERZIJA: * 1. Svaki stupac za svaki dan spaja točno svoja 2 retka (ne spaja stupce međusobno). * 2. batchUpdate šalje sve u jednom ili dva velika bloka (nema timeouta). * 3. Format u redu zbroja odgovara točno originalnom obrascu knjigovodstva. */ function generirajIzvjestajZaKnjigovodstvo_(month) { const parts = month.split("-"); const year = parseInt(parts[0], 10); const monthIndex = parseInt(parts[1], 10) - 1; const daysInMonth = new Date(year, monthIndex + 1, 0).getDate(); const tz = SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(); const sourceSheet = getSheet_(); const raw = sourceSheet.getDataRange().getValues(); const headers = raw.shift(); const byDateWorker = {}; raw.forEach(function (row) { const obj = {}; headers.forEach(function (h, i) { obj[h] = normalizeCell_(h, row[i], tz); }); if (!obj["Datum"] || !obj["Zaposlenik"]) return; byDateWorker[obj["Datum"] + "|" + obj["Zaposlenik"]] = obj; }); const holidaySet = hrvatskiBlagdani_(year); const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheetName = "Izvoz " + month; let outSheet = ss.getSheetByName(sheetName); if (outSheet) ss.deleteSheet(outSheet); outSheet = ss.insertSheet(sheetName); const gid = outSheet.getSheetId(); // Radni dani (pon-pet) const weekdays = []; for (let d = 1; d <= daysInMonth; d++) { const dateObj = new Date(year, monthIndex, d); const dow = dateObj.getDay(); if (dow === 0 || dow === 6) continue; const dateStr = Utilities.formatDate(dateObj, tz, "yyyy-MM-dd"); const dateDisplay = Utilities.formatDate(dateObj, tz, "dd.MM.yyyy"); weekdays.push({ dateObj: dateObj, dateStr: dateStr, dateDisplay: dateDisplay, isHoliday: holidaySet.has(dateStr) }); } const grid = []; const merges = []; const reviewRows = []; const employeeBlocks = []; // --- 1. Zaglavlje (2 retka) --- const hdr1 = blankRow40_(); const hdr2 = blankRow40_(); hdr1[0] = "Datum"; hdr1[2] = "Radno mjesto"; hdr1[3] = "Od - do"; hdr2[3] = "Od - do"; FIELD_COLS_.forEach(function (f) { hdr1[f.col - 1] = f.header; }); grid.push(hdr1, hdr2); merges.push([0, 2, 0, 1]); // Datum (A1:A2) merges.push([0, 2, 1, 2]); // Prazno/Datum kolona (B1:B2) merges.push([0, 2, 2, 3]); // Radno mjesto (C1:C2) FIELD_COLS_.forEach(function (f) { merges.push([0, 2, f.col - 1, f.col]); }); let rowCursor = 2; // --- 2. Radnici i dani --- WORKERS_ORDER_.forEach(function (worker) { const info = EMPLOYEE_CODES_[worker]; if (!info) return; const codeStr = String(info.code); const padded = codeStr.length >= 6 ? codeStr : (new Array(6 - codeStr.length + 1).join(" ") + codeStr); const empHeaderRow = blankRow40_(); empHeaderRow[0] = "-"; empHeaderRow[1] = padded + " - " + info.display; grid.push(empHeaderRow); const empHeaderRow0 = rowCursor; rowCursor++; merges.push([empHeaderRow0, empHeaderRow0 + 1, 1, 40]); // B:AN spajanje za ime radnika const dataStart0 = rowCursor; const sched = WORKER_SCHEDULE_[worker] || { dolazak: "08:00", odlazak: "16:00" }; const dnevnaNorma = satiIzmedju_(sched.dolazak, sched.odlazak); const totals = {}; FIELD_COLS_.forEach(function (f) { totals[f.col] = 0; }); weekdays.forEach(function (wd) { const rowA = blankRow40_(); const rowB = blankRow40_(); rowA[1] = wd.dateObj; rowA[2] = RADNO_MJESTO_[worker] || ""; rowA[7] = dnevnaNorma / 24; // Satnica rowA[21] = 24 - dnevnaNorma; // Dnevni odmor const rec = byDateWorker[wd.dateStr + "|" + worker]; let needsReview = false; if (rec && rec["Dolazak"] && rec["Odlazak"] && rec["Dolazak"] !== "" && rec["Odlazak"] !== "") { rowA[3] = wd.dateDisplay + " " + rec["Dolazak"] + " - " + wd.dateDisplay + " " + rec["Odlazak"]; const sati = satiIzmedju_(rec["Dolazak"], rec["Odlazak"]); rowA[4] = sati / 24; totals[5] += sati; } else if (rec && rec["Izostanak"]) { rowA[3] = wd.dateDisplay + " 00:00 - " + wd.dateDisplay + " 00:00"; const kolona = mapirajIzostanak_(rec["Izostanak"]); if (kolona) { const f = FIELD_COLS_.find(function (x) { return x.header === kolona; }); if (f) { rowA[f.col - 1] = dnevnaNorma / 24; totals[f.col] += dnevnaNorma; } } } else if (wd.isHoliday) { rowA[3] = wd.dateDisplay + " 00:00 - " + wd.dateDisplay + " 00:00"; rowA[15] = dnevnaNorma / 24; totals[16] += dnevnaNorma; } else { rowA[3] = ""; needsReview = true; } rowB[3] = " - "; totals[8] += dnevnaNorma; totals[22] += (24 - dnevnaNorma); const r0 = rowCursor; grid.push(rowA, rowB); rowCursor += 2; // ISPRAVAK MERGE-a: Spaja se točno 2 vertikalna retka ZA SVAKI STUPAC pojedinačno // Stupci A (0), B (1), C (2) merges.push([r0, r0 + 2, 0, 1]); merges.push([r0, r0 + 2, 1, 2]); merges.push([r0, r0 + 2, 2, 3]); // Stupci E do AN (indeksi 4 do 39) - stupac D (indeks 3) ostaje odvojen za "Od - do" i " - " for (let c = 4; c < 40; c++) { merges.push([r0, r0 + 2, c, c + 1]); } if (needsReview) reviewRows.push(r0); }); const dataEndExcl0 = rowCursor; // Redak sa zbrojem (Totals) const totalsRow = blankRow40_(); FIELD_COLS_.forEach(function (f) { const val = totals[f.col] || 0; // U originalu knjigovodstva zbrojevi su obični brojevi/stringovi bez dijeljenja s 24 totalsRow[f.col - 1] = val === 0 ? "0" : String(Math.round(val * 100) / 100); }); grid.push(totalsRow); const totalsRow0 = rowCursor; rowCursor++; employeeBlocks.push({ empHeaderRow0: empHeaderRow0, dataStart0: dataStart0, dataEndExcl0: dataEndExcl0, totalsRow0: totalsRow0 }); }); // --- 3. Upis podataka u Sheet --- outSheet.getRange(1, 1, grid.length, 40).setValues(grid); // --- 4. Formatiranje --- const fullRange = outSheet.getRange(1, 1, grid.length, 40); fullRange.setFontFamily("Tahoma").setFontSize(8).setVerticalAlignment("top"); fullRange.setBorder(true, true, true, true, true, true, "#a0a0a0", SpreadsheetApp.BorderStyle.SOLID); fullRange.setBackground("#ffffff"); outSheet.getRange(1, 1, grid.length, 1).setHorizontalAlignment("left"); outSheet.getRange(1, 2, grid.length, 1).setNumberFormat("yyyy-mm-dd").setHorizontalAlignment("center"); outSheet.getRange(1, 3, grid.length, 1).setHorizontalAlignment("left"); outSheet.getRange(1, 4, grid.length, 1).setHorizontalAlignment("center"); FIELD_COLS_.forEach(function (f) { const rng = outSheet.getRange(1, f.col, grid.length, 1); rng.setHorizontalAlignment("center"); if (f.type === "hours") rng.setNumberFormat("h:mm"); }); outSheet.getRange(1, 1, 2, 40).setBackground("#b0cbf0"); employeeBlocks.forEach(function (b) { outSheet.getRange(b.empHeaderRow0 + 1, 1, 1, 40).setBackground("#fffdca"); outSheet.getRange(b.empHeaderRow0 + 1, 1, 1, 1).setBackground("#f0f0f0"); outSheet.getRange(b.dataStart0 + 1, 1, b.dataEndExcl0 - b.dataStart0, 1).setBackground("#fffdca"); // Zbrojevi const totRange = outSheet.getRange(b.totalsRow0 + 1, 1, 1, 40); totRange.setBackground("#ffd895"); totRange.setNumberFormat("@").setHorizontalAlignment("right"); }); reviewRows.forEach(function (r0) { outSheet.getRange(r0 + 1, 1, 2, 40).setBackground("#fff3cd"); outSheet.getRange(r0 + 1, 4).setNote("Nema zabilježenog klika u aplikaciji za ovaj dan."); }); for (let c = 1; c <= 40; c++) { const excelW = COL_WIDTHS_EXCEL_[c] || 8; outSheet.setColumnWidth(c, Math.round(excelW * 7 + 5)); } outSheet.setRowHeights(1, grid.length, 18); outSheet.setFrozenRows(2); // --- 5. Brzo grupno spajanje (Sheets API) --- spojiCelijeOptimizirano_(ss.getId(), gid, merges); return "https://docs.google.com/spreadsheets/d/" + ss.getId() + "/export?format=xlsx&gid=" + gid; } /** * Šalje do 5000 spajanja po pozivu čime se izbjegava timeout (2 poziva umjesto 20). */ function spojiCelijeOptimizirano_(spreadsheetId, gid, merges) { const CHUNK = 5000; for (let i = 0; i < merges.length; i += CHUNK) { const chunk = merges.slice(i, i + CHUNK); const requests = chunk.map(function (m) { return { mergeCells: { range: { sheetId: gid, startRowIndex: m[0], endRowIndex: m[1], startColumnIndex: m[2], endColumnIndex: m[3] }, mergeType: "MERGE_ALL" } }; }); Sheets.Spreadsheets.batchUpdate({ requests: requests }, spreadsheetId); } }
